📜 DDL PostgreSQL & Agregasi Multi-Tenant (PRD Seksi 9 & 10)

Production Schema 
-- 1. Tenants Table dengan Gating Paket & Kuota
CREATE TABLE tenants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    slug VARCHAR(50) UNIQUE NOT NULL,
    owner_phone VARCHAR(20) NOT NULL,
    timezone VARCHAR(50) DEFAULT 'Asia/Jakarta',
    subscription_plan VARCHAR(20) DEFAULT 'STARTER', -- STARTER, PRO, CLINIC, LIFETIME_PARTNER
    subscription_until TIMESTAMPTZ NOT NULL,
    is_accepting_patients BOOLEAN DEFAULT true,
    reschedule_cutoff_hours INT DEFAULT 2,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Anti-Tumpang Tindih Jadwal (PostgreSQL btree_gist Range-Lock)
CREATE TABLE appointments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
    service_id UUID NOT NULL REFERENCES services(id),
    customer_phone VARCHAR(20) NOT NULL,
    customer_name VARCHAR(100) NOT NULL,
    scheduled_time TSTZRANGE NOT NULL,
    status VARCHAR(30) DEFAULT 'CONFIRMED',
    EXCLUDE USING gist (
        tenant_id WITH =,
        scheduled_time WITH &&
    ) WHERE (status NOT IN ('CANCELLED'))
);

-- 3. View Monitoring Utilisasi Kuota Bulanan Tenant (PRD 10.1)
CREATE OR REPLACE VIEW view_tenant_quota_monitoring AS
SELECT 
    t.id AS tenant_id,
    t.name AS tenant_name,
    t.slug,
    t.subscription_plan,
    t.subscription_until,
    COUNT(a.id) AS current_month_bookings,
    CASE 
        WHEN t.subscription_plan = 'STARTER' THEN 100
        WHEN t.subscription_plan = 'PRO' THEN 400
        WHEN t.subscription_plan = 'LIFETIME_PARTNER' THEN 250
        ELSE 999999 
    END AS max_allowed_quota
FROM tenants t
LEFT JOIN appointments a ON t.id = a.tenant_id 
    AND a.created_at >= DATE_TRUNC('month', NOW())
    AND a.status NOT IN ('CANCELLED')
GROUP BY t.id, t.name, t.slug, t.subscription_plan, t.subscription_until;
